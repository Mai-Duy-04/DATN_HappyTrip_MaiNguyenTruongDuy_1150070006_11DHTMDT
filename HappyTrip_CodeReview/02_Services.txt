using Microsoft.EntityFrameworkCore;
using TourWebApp.Data.Models;
using TourWebApp.Models;

namespace TourWebApp.Services;

public class CodOrderExpiryService : BackgroundService
{
    private readonly IServiceScopeFactory _scopeFactory;
    private readonly ILogger<CodOrderExpiryService> _logger;

    public CodOrderExpiryService(IServiceScopeFactory scopeFactory, ILogger<CodOrderExpiryService> logger)
    {
        _scopeFactory = scopeFactory;
        _logger = logger;
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        try
        {
            using var initialScope = _scopeFactory.CreateScope();
            var inventory = initialScope.ServiceProvider.GetRequiredService<ISeatInventoryService>();
            await inventory.SynchronizeAllAsync(stoppingToken);
        }
        catch (Exception ex) when (!stoppingToken.IsCancellationRequested)
        {
            _logger.LogError(ex, "Initial seat inventory synchronization failed.");
        }

        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                await XuLyDonQuaHan(stoppingToken);
                await Task.Delay(TimeSpan.FromMinutes(1), stoppingToken);
            }
            catch (OperationCanceledException) when (stoppingToken.IsCancellationRequested)
            {
                break;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Booking expiry job failed.");
            }
        }
    }

    private async Task XuLyDonQuaHan(CancellationToken stoppingToken)
    {
        using var scope = _scopeFactory.CreateScope();
        var db = scope.ServiceProvider.GetRequiredService<ApplicationDbContext>();
        var seatInventory = scope.ServiceProvider.GetRequiredService<ISeatInventoryService>();
        var now = DateTime.Now;

        var donQuaHan = await db.DonDatTours
            .Include(x => x.IdLichNavigation)
            .Include(x => x.IdTourNavigation)
            .Include(x => x.PhieuGiamGiaSuDungs)
            .Where(x => !x.DaThanhToan
                && x.TrangThai != BookingPaymentStatus.TrangThaiDaHuy
                && x.HanThanhToan.HasValue
                && x.HanThanhToan.Value < now)
            .ToListAsync(stoppingToken);

        if (donQuaHan.Count == 0) return;

        foreach (var don in donQuaHan)
        {
            don.TrangThai = BookingPaymentStatus.TrangThaiDaHuy;
            don.TrangThaiThanhToan = BookingPaymentStatus.TrangThaiTtHetHanThanhToan;

            var suDung = don.PhieuGiamGiaSuDungs.OrderByDescending(x => x.IdSuDung).FirstOrDefault();
            if (suDung != null && suDung.TrangThai == "GiuCho")
            {
                suDung.TrangThai = "DaHuy";
                suDung.ThoiDiemSuDung = now;
                suDung.GhiChu = "Don COD het han tu dong";
            }
        }

        var affectedSchedules = donQuaHan
            .Select(x => new { x.IdLich, x.IdTour })
            .Distinct()
            .ToList();

        await db.SaveChangesAsync(stoppingToken);
        foreach (var item in affectedSchedules)
        {
            await seatInventory.SynchronizeAsync(item.IdLich, item.IdTour, stoppingToken);
        }
        _logger.LogInformation("Booking expiry job canceled {Count} expired orders.", donQuaHan.Count);
    }
}

    private async Task SendEmailInternalAsync(
        string toEmail,
        string subject,
        string plainTextBody,
        string? htmlBody = null,
        bool includeBrandImages = false)
    {
        ValidateSmtpSettings();

        try
        {
            using var client = new SmtpClient(
                _settings.SmtpHost,
                _settings.SmtpPort)
            {
                EnableSsl = _settings.EnableSsl,
                UseDefaultCredentials = false,
                Credentials = new NetworkCredential(
                    _settings.Username,
                    _settings.Password),
                DeliveryMethod = SmtpDeliveryMethod.Network,
                Timeout = 30000
            };

            using var message = new MailMessage
            {
                From = new MailAddress(
                    _settings.SenderEmail,
                    _settings.SenderName),
                Subject = subject,
                SubjectEncoding = Encoding.UTF8,
                BodyEncoding = Encoding.UTF8,
                Body = plainTextBody,
                IsBodyHtml = false
            };

            message.To.Add(new MailAddress(toEmail));

            if (!string.IsNullOrWhiteSpace(htmlBody))
            {
                var textView =
                    AlternateView.CreateAlternateViewFromString(
                        plainTextBody,
                        Encoding.UTF8,
                        MediaTypeNames.Text.Plain);

                var htmlView =
                    AlternateView.CreateAlternateViewFromString(
                        htmlBody,
                        Encoding.UTF8,
                        MediaTypeNames.Text.Html);

                if (includeBrandImages)
                {
                    var webRoot = _environment.WebRootPath;

                    if (string.IsNullOrWhiteSpace(webRoot))
                    {
                        throw new InvalidOperationException(
                            "Không tìm thấy thư mục wwwroot.");
                    }

                    var folder = Path.Combine(
                        webRoot,
                        "img",
                        "email");

                    AddImage(
                        htmlView,
                        Path.Combine(
                            folder,
                            "happytrip-email-header.png"),
                        "happytrip-header");

                    AddImage(
                        htmlView,
                        Path.Combine(
                            folder,
                            "happytrip-email-footer.png"),
                        "happytrip-footer");
                }

                message.AlternateViews.Add(textView);
                message.AlternateViews.Add(htmlView);
            }

            await client.SendMailAsync(message);
        }
        catch (SmtpException ex)
        {
            _logger.LogError(
                ex,
                "Gui email qua SMTP that bai.");

            throw new InvalidOperationException(
                "Không thể gửi email. Vui lòng kiểm tra cấu hình SMTP và thử lại.",
                ex);
        }
        catch (Exception ex)
        {
            _logger.LogError(
                ex,
                "Xay ra loi khi gui email.");

            throw;
        }
    }

    private static void AddImage(
        AlternateView htmlView,
        string imagePath,
        string contentId)
    {
        if (!File.Exists(imagePath))
        {
            throw new FileNotFoundException(
                $"Không tìm thấy ảnh HappyTrip: {imagePath}");
        }

        var resource = new LinkedResource(
            imagePath,
            MediaTypeNames.Image.Png)
        {
            ContentId = contentId,
            TransferEncoding = TransferEncoding.Base64
        };

        htmlView.LinkedResources.Add(resource);
    }

    // ==========================================
    // 6. KIEM TRA SMTP
    // ==========================================

    private void ValidateSmtpSettings()
    {
        if (string.IsNullOrWhiteSpace(_settings.SmtpHost) ||
            string.IsNullOrWhiteSpace(_settings.SenderEmail) ||
            string.IsNullOrWhiteSpace(_settings.Username) ||
            string.IsNullOrWhiteSpace(_settings.Password))
        {
            throw new InvalidOperationException(
                "Chưa cấu hình đầy đủ SMTP cho HappyTrip.");
        }

        if (_settings.SmtpPort < 1 ||
            _settings.SmtpPort > 65535)
        {
            throw new InvalidOperationException(
                "Cổng SMTP không hợp lệ.");
        }
    }

    // ========================================================
    // 1. PHAN TICH NHU CAU DU LICH
    // ========================================================

    public async Task<GeminiTravelResult> PhanTichAsync(
        GeminiTravelRequest request,
        CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(request);

        if (string.IsNullOrWhiteSpace(request.NoiDung))
        {
            throw new ArgumentException(
                "Nội dung phân tích nhu cầu không được trống.");
        }

        if (request.NoiDung.Length > MaxInputCharacters)
        {
            throw new ArgumentException(
                "Nội dung phân tích quá dài.");
        }

        string apiKey =
            _configuration["Gemini:ApiKey"] ?? "";

        if (string.IsNullOrWhiteSpace(apiKey))
        {
            throw new InvalidOperationException(
                "Chưa cấu hình Gemini API key.");
        }

        string primaryModel =
            _configuration["Gemini:ExtractionModel"]
            ?? _configuration["Gemini:Model"]
            ?? "gemini-2.5-flash";

        string fallbackModel =
            _configuration["Gemini:ExtractionFallbackModel"]
            ?? _configuration["Gemini:FallbackModel"]
            ?? "gemini-2.5-flash-lite";

        primaryModel = primaryModel.Trim();
        fallbackModel = fallbackModel.Trim();

        if (string.IsNullOrWhiteSpace(primaryModel))
        {
            primaryModel = "gemini-2.5-flash";
        }

        if (string.Equals(
            primaryModel,
            fallbackModel,
            StringComparison.OrdinalIgnoreCase))
        {
            fallbackModel = "";
        }

        List<string> destinations =
            CleanCatalog(request.DiemDens);

        List<string> interests =
            CleanCatalog(request.SoThichs);

        List<string> tourTypes =
            CleanCatalog(request.LoaiTours);

        string today =
            DateTimeOffset.UtcNow
                .ToOffset(TimeSpan.FromHours(7))
                .ToString(
                    "yyyy-MM-dd",
                    CultureInfo.InvariantCulture);

    private static List<string> CleanCatalog(
        List<string>? items)
    {
        return (items ?? new List<string>())
            .Where(x =>
                !string.IsNullOrWhiteSpace(x))
            .Select(x => x.Trim())
            .Distinct(

    private static string? FindInCatalog(
        List<string> catalog,
        string? value)
    {
        if (string.IsNullOrWhiteSpace(value))
        {
            return null;
        }

        return catalog.FirstOrDefault(x =>
            string.Equals(
                x,
                value.Trim(),
                StringComparison.OrdinalIgnoreCase));

    private static string? ReadString(

        string name)

        return root.TryGetProperty(

                ? value.GetString()?.Trim()

    private static List<string> ReadStringArray(
        JsonElement root,
        string name)
    {
        if (!root.TryGetProperty(
                name,
                out JsonElement value) ||
            value.ValueKind != JsonValueKind.Array)
        {
            return new List<string>();
        }

        return value.EnumerateArray()
            .Where(x =>
                x.ValueKind == JsonValueKind.String)
            .Select(x =>
                x.GetString()?.Trim() ?? "")
            .Where(x =>
                x.Length > 0)
            .ToList();
    }

        string name)

                name,
                out JsonElement value) &&
            value.ValueKind == JsonValueKind.Number &&
            value.TryGetDecimal(out decimal number)
                ? number
                : null;
    }

    private static int? ReadInt(
        JsonElement root,
        string name)

        return root.TryGetProperty(
                name,
                out JsonElement value) &&
            value.ValueKind == JsonValueKind.Number &&
            value.TryGetInt32(out int number)

    private static async Task<JsonDocument>
        SendWithFallbackAsync(
            string primaryModel,
            string fallbackModel,
            string apiKey,
            object payload,
            CancellationToken cancellationToken)
    {
        try
        {
            return await SendToModelAsync(
                primaryModel,
                apiKey,
                payload,
                PrimaryAttempts,
                cancellationToken);
        }
        catch (GeminiApiException primaryError)
            when (
                !string.IsNullOrWhiteSpace(fallbackModel) &&
                IsTemporaryServiceError(primaryError.StatusCode))
        {
            try
            {
                return await SendToModelAsync(
                    fallbackModel,
                    apiKey,
                    payload,
                    FallbackAttempts,
                    cancellationToken);
            }
            catch (GeminiApiException fallbackError)
            {
                throw new InvalidOperationException(
                    $"Gemini chính ({primaryModel}) " +
                    $"lỗi HTTP {(int)primaryError.StatusCode}; " +
                    $"Gemini dự phòng ({fallbackModel}) " +
                    $"lỗi HTTP {(int)fallbackError.StatusCode}: " +
                    fallbackError.Message,
                    fallbackError);
            }
        }
        catch (GeminiApiException ex)
        {
            throw new InvalidOperationException(
                $"Gemini API lỗi HTTP {(int)ex.StatusCode}: " +
                ex.Message,
                ex);
        }
    }

    // ========================================================
    // 12. SEND TO MODEL WITH RETRY
    // ========================================================

    private static async Task<JsonDocument>
        SendToModelAsync(
            string model,
            string apiKey,
            object payload,
            int maxAttempts,
            CancellationToken cancellationToken)
    {
        string endpoint =
            "https://generativelanguage.googleapis.com/" +
            "v1beta/models/" +
            Uri.EscapeDataString(model) +
            ":generateContent";

        for (int attempt = 1;
             attempt <= maxAttempts;
             attempt++)
        {
            cancellationToken.ThrowIfCancellationRequested();

            using var request =
                new HttpRequestMessage(
                    HttpMethod.Post,
                    endpoint);

            request.Headers.Add(
                "x-goog-api-key",
                apiKey);

            request.Content =
                JsonContent.Create(payload);

            HttpResponseMessage response;

            try
            {
                response = await Client.SendAsync(
                    request,
                    cancellationToken);
            }
            catch (HttpRequestException)
                when (attempt < maxAttempts)
            {
                await Task.Delay(
                    GetRetryDelay(attempt, null),
                    cancellationToken);

                continue;
            }

            using (response)
            {
                if (response.IsSuccessStatusCode)
                {
                    await using Stream stream =
                        await response.Content.ReadAsStreamAsync(
                            cancellationToken);

                    return await JsonDocument.ParseAsync(
                        stream,
                        cancellationToken:
                            cancellationToken);
                }

                bool retryable =
                    response.StatusCode is
                        HttpStatusCode.TooManyRequests or
                        HttpStatusCode.InternalServerError or
                        HttpStatusCode.BadGateway or
                        HttpStatusCode.ServiceUnavailable or
                        HttpStatusCode.GatewayTimeout;

                if (retryable &&
                    attempt < maxAttempts)
                {
                    await Task.Delay(
                        GetRetryDelay(attempt, response),
                        cancellationToken);

                    continue;
                }

                string errorBody =
                    await response.Content.ReadAsStringAsync(
                        cancellationToken);

                throw new GeminiApiException(
                    response.StatusCode,
                    GetGoogleErrorMessage(errorBody));
            }
        }

        throw new InvalidOperationException(
            "Không thể kết nối Gemini sau các lần thử.");
    }

    // ========================================================
    // 13. TEMPORARY ERRORS
    // ========================================================

    private static bool IsTemporaryServiceError(
        HttpStatusCode code)
    {
        return code is
            HttpStatusCode.InternalServerError or
            HttpStatusCode.BadGateway or
            HttpStatusCode.ServiceUnavailable or
            HttpStatusCode.GatewayTimeout;
    }

    // ========================================================
    // 14. EXPONENTIAL BACKOFF
    // ========================================================

    private static TimeSpan GetRetryDelay(
        int attempt,
        HttpResponseMessage? response)
    {
        double seconds =
            Math.Min(
                Math.Pow(2, attempt - 1),
                8);

        var retryAfter =
            response?.Headers.RetryAfter;

        if (retryAfter?.Delta is TimeSpan delta &&
            delta > TimeSpan.Zero)
        {
            seconds = Math.Clamp(
                delta.TotalSeconds,
                1,
                12);
        }
        else if (
            retryAfter?.Date is DateTimeOffset date &&
            date > DateTimeOffset.UtcNow)
        {
            seconds = Math.Clamp(
                (date - DateTimeOffset.UtcNow)
                    .TotalSeconds,
                1,
                12);
        }

        return TimeSpan.FromSeconds(
            seconds +
            Random.Shared.Next(100, 501) / 1000.0);
    }

    // ========================================================
    // 15. GOOGLE ERROR MESSAGE
    // ========================================================

    private static string GetGoogleErrorMessage(
        string body)
    {
        try
        {
            using JsonDocument parsed =
                JsonDocument.Parse(body);

            if (parsed.RootElement.TryGetProperty(
                    "error",
                    out JsonElement error) &&
                error.TryGetProperty(
                    "message",
                    out JsonElement message) &&
                message.ValueKind ==
                    JsonValueKind.String)
            {
                string? text = message.GetString();

                if (!string.IsNullOrWhiteSpace(text))
                {
                    return text;
                }
            }
        }
        catch (JsonException)
        {
            // Google co the tra loi khong phai JSON.
        }

        return "Không có thông tin chi tiết từ Google.";
    }

// ============================================================
// MATCHING SCORE SERVICE
// ============================================================

public sealed class MatchingScoreService : IMatchingScoreService
{
    private readonly ApplicationDbContext _db;

    private const int DiemToiDaSoThich = 50;
    private const int DiemToiDaDiemDen = 30;
    private const int DiemToiDaNganSach = 20;

    public MatchingScoreService(ApplicationDbContext db)
    {
        _db = db;
    }

    // ========================================================
    // 1. HAM GOI Y TOUR
    // ========================================================

    public async Task<List<MatchingScoreResult>> GoiYTourAsync(
        MatchingScoreRequest request,
        CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(request);

        // ====================================================
        // 2. VALIDATE
        // ====================================================

        if (request.IdDiemDen is <= 0)
        {
            throw new ArgumentException(
                "Điểm đến không hợp lệ.");
        }

        if (request.NganSachToiDa is <= 0)
        {
            throw new ArgumentException(
                "Ngân sách phải lớn hơn 0.");
        }

        if (request.SoNgayToiDa is <= 0)
        {
            throw new ArgumentException(
                "Số ngày phải lớn hơn 0.");
        }

        if (request.SoKhach is <= 0)
        {
            throw new ArgumentException(
                "Số khách phải lớn hơn 0.");
        }

        int topN = Math.Clamp(
            request.TopN,
            1,
            20);

        HashSet<int> soThichDaChon =
            (request.IdSoThich ?? new List<int>())
                .Where(id => id > 0)
                .ToHashSet();

        bool coSoThich =
            soThichDaChon.Count > 0;

        bool coLoaiTour =
            !string.IsNullOrWhiteSpace(request.LoaiTour);

        bool coNgay =
            request.NgayDuKien.HasValue;

        bool coSoKhach =
            request.SoKhach.HasValue;

        // Chi yeu cau lich thuc te khi can xac nhan
        // ngay hoac so cho.
        //
        // Neu khach chi hoi tim tour, he thong van
        // cho phep kham pha danh muc tour.
        // Tuy nhien KHONG khang dinh tour co lich
        // trong tuong lai khi chua kiem tra.

        bool canLocTheoLich =
            coNgay;

        DateOnly homNay = DateOnly.FromDateTime(
            DateTime.UtcNow.AddHours(7));

        // Khong co lich khoi hanh qua khu.

        if (request.NgayDuKien.HasValue &&
            request.NgayDuKien.Value < homNay)
        {
            return new List<MatchingScoreResult>();
        }

        string loaiNganSach =
            NormalizeText(request.LoaiNganSach);

        bool nganSachCaNhom =
            loaiNganSach == "canhom";

        // Neu ngan sach la cua ca nhom,
        // can biet so nguoi moi co the
        // uoc tinh tong chi phi.
        //
        // Khong tu gia dinh nhom co bao nhieu nguoi.

        if (nganSachCaNhom &&
            request.NganSachToiDa.HasValue &&
            !request.SoKhach.HasValue)
        {
            return new List<MatchingScoreResult>();
        }

        string? loaiTourYeuCau =
            request.LoaiTour?.Trim();

        // ====================================================
        // 3. LOAD DU LIEU
        // ====================================================

        var tours =
            new Dictionary<int, TourMatchingData>();

        DbConnection connection =
            _db.Database.GetDbConnection();

        bool openedHere =
            connection.State != ConnectionState.Open;

        if (openedHere)
        {
            await connection.OpenAsync(cancellationToken);
        }

        try
        {
            // ================================================
            // 3.1. TOUR + DIEM DEN + LOAI TOUR
            // ================================================

            await using (DbCommand command =
                connection.CreateCommand())
            {
                command.CommandText = """
                    SELECT
                        t.IdTour,
                        t.TenTour,
                        t.DiaDiem,
                        t.HinhAnh,
                        t.ThoiGian,
                        COALESCE(
                            NULLIF(t.GiaKhuyenMai, 0),
                            NULLIF(t.GiaGoc, 0)
                        ) AS GiaBan,
                        td.IdDiemDen,
                        td.VaiTro,
                        t.IdLoaiTour,
                        lt.TenLoai
                    FROM dbo.Tour AS t
                    LEFT JOIN dbo.AITourDiemDen AS td
                        ON td.IdTour = t.IdTour
                    LEFT JOIN dbo.LoaiTour AS lt
                        ON lt.IdLoaiTour = t.IdLoaiTour
                    WHERE t.TrangThai = 1
                      AND COALESCE(
                          NULLIF(t.GiaKhuyenMai, 0),
                          NULLIF(t.GiaGoc, 0)
                      ) > 0
                    ORDER BY t.IdTour;
                    """;

                await using DbDataReader reader =
                    await command.ExecuteReaderAsync(
                        cancellationToken);

                while (await reader.ReadAsync(
                    cancellationToken))
                {
                    int idTour =
                        reader.GetInt32(0);

                    if (!tours.TryGetValue(
                        idTour,
                        out TourMatchingData? tour))
                    {
                        string thoiGian =
                            ReadText(reader, 4);

                        tour = new TourMatchingData
                        {
                            IdTour = idTour,

                            TenTour =
                                ReadText(reader, 1),

                            DiaDiem =
                                ReadText(reader, 2),

                            HinhAnh =
                                ReadText(reader, 3),

                            ThoiGian = thoiGian,

                            SoNgay =
                                ParseSoNgay(thoiGian),

                            GiaBan =
                                reader.GetDecimal(5),

                            IdLoaiTour =
                                reader.IsDBNull(8)
                                    ? 0
                                    : reader.GetInt32(8),

                            TenLoaiTour =
                                ReadText(reader, 9)
                        };

                        tours.Add(idTour, tour);
                    }

                    if (!reader.IsDBNull(6))
                    {
                        string vaiTro =
                            ReadText(reader, 7);

                        if (IsDiemDenRole(vaiTro))
                        {
                            tour.DiemDenIds.Add(
                                reader.GetInt32(6));
                        }
                    }
                }
            }

            // ================================================
            // 3.2. TOUR + SO THICH
            // ================================================

            await using (DbCommand command =
                connection.CreateCommand())
            {
                command.CommandText = """
                    SELECT
                        ts.IdTour,
                        ts.IdSoThich
                    FROM dbo.AITourSoThich AS ts
                    INNER JOIN dbo.Tour AS t
                        ON t.IdTour = ts.IdTour
                    WHERE t.TrangThai = 1;
                    """;

                await using DbDataReader reader =
                    await command.ExecuteReaderAsync(
                        cancellationToken);

                while (await reader.ReadAsync(
                    cancellationToken))
                {
                    int idTour =
                        reader.GetInt32(0);

                    int idSoThich =
                        reader.GetInt32(1);

                    if (tours.TryGetValue(
                        idTour,
                        out TourMatchingData? tour))
                    {
                        tour.SoThichIds.Add(idSoThich);
                    }
                }
            }

            // ================================================
            // 3.3. LICH KHOI HANH
            // ================================================

            if (canLocTheoLich)
            {
                await using DbCommand command =
                    connection.CreateCommand();

                command.CommandText = """
                    SELECT
                        l.IdTour,
                        l.IdLich,
                        l.NgayKhoiHanh,
                        l.SoChoToiDa,
                        l.SoChoConLai,
                        l.TrangThai
                    FROM dbo.LichKhoiHanh AS l
                    INNER JOIN dbo.Tour AS t
                        ON t.IdTour = l.IdTour
                    WHERE t.TrangThai = 1
                      AND l.SoChoToiDa > 0
                      AND l.SoChoConLai > 0
                      AND l.SoChoConLai <= l.SoChoToiDa
                      AND l.NgayKhoiHanh >= @NgayHomNay
                      AND l.TrangThai IN (
                          N'Đang mở',
                          N'Còn chỗ',
                          N'Sắp hết chỗ'
                      )
                    ORDER BY
                        l.IdTour,
                        l.NgayKhoiHanh,
                        l.IdLich;
                    """;

                DbParameter parameter =
                    command.CreateParameter();

                parameter.ParameterName =
                    "@NgayHomNay";

                parameter.DbType =
                    DbType.Date;

                parameter.Value =
                    homNay.ToDateTime(TimeOnly.MinValue);

                command.Parameters.Add(parameter);

                await using DbDataReader reader =
                    await command.ExecuteReaderAsync(
                        cancellationToken);

                while (await reader.ReadAsync(
                    cancellationToken))
                {
                    int idTour =
                        reader.GetInt32(0);

                    if (!tours.TryGetValue(
                        idTour,
                        out TourMatchingData? tour))
                    {
                        continue;
                    }

                    DateOnly? ngayKhoiHanh =
                        ReadDateOnly(reader, 2);

                    if (!ngayKhoiHanh.HasValue)
                    {
                        continue;
                    }

                    if (ngayKhoiHanh.Value < homNay)
                    {
                        continue;
                    }

                    int soChoToiDa =
                        reader.GetInt32(3);

                    int soChoConLai =
                        reader.GetInt32(4);

                    if (soChoToiDa <= 0 ||
                        soChoConLai <= 0 ||
                        soChoConLai > soChoToiDa)
                    {
                        continue;
                    }

                    tour.LichKhoiHanhs.Add(
                        new LichMatchingData
                        {
                            IdLich =
                                reader.GetInt32(1),

                            NgayKhoiHanh =
                                ngayKhoiHanh.Value,

                            SoChoConLai =
                                soChoConLai,

                            TrangThai =
                                ReadText(reader, 5)
                        });
                }
            }
        }
        finally
        {
            if (openedHere)
            {
                await connection.CloseAsync();
            }
        }

        // ====================================================
        // 4. FILTER + MATCHING SCORE
        // ====================================================

        var results =
            new List<MatchingScoreResult>();

        foreach (TourMatchingData tour in tours.Values)
        {
            // ================================================
            // 4.1. DIEM DEN
            // ================================================

            if (request.IdDiemDen.HasValue &&
                !tour.DiemDenIds.Contains(
                    request.IdDiemDen.Value))
            {
                continue;
            }

            // ================================================
            // 4.2. NGAN SACH
            // ================================================

            if (request.NganSachToiDa.HasValue)
            {
                decimal giaCanSoSanh =
                    tour.GiaBan;

                if (nganSachCaNhom)
                {
                    // Gia tour dang dung la gia mot nguoi.
                    // Uoc tinh nhom theo so khach.
                    //
                    // Luu y: neu nhom co tre em,
                    // can lay them gia theo do tuoi
                    // tu nghiep vu dat tour de tinh chinh xac.

                    int soKhach =
                        request.SoKhach!.Value;

                    giaCanSoSanh =
                        tour.GiaBan * soKhach;
                }

                if (giaCanSoSanh >
                    request.NganSachToiDa.Value)
                {
                    continue;
                }
            }

            // ================================================
            // 4.3. SO NGAY
            // ================================================

            if (request.SoNgayToiDa.HasValue)
            {
                if (!tour.SoNgay.HasValue ||
                    tour.SoNgay.Value >
                        request.SoNgayToiDa.Value)
                {
                    continue;
                }
            }

            // ================================================
            // 4.4. SO THICH
            // ================================================

            int soThichTrung =
                tour.SoThichIds.Count(
                    id => soThichDaChon.Contains(id));

            if (coSoThich &&
                soThichTrung == 0)
            {
                continue;
            }

            // ================================================
            // 4.5. LOAI TOUR
            // ================================================

            if (coLoaiTour &&
                !string.Equals(
                    NormalizeText(tour.TenLoaiTour),
                    NormalizeText(loaiTourYeuCau),
                    StringComparison.Ordinal))
            {
                continue;
            }

            // ================================================
            // 4.6. LICH + SO KHACH
            // ================================================

            LichMatchingData? lichPhuHop = null;

            if (canLocTheoLich)
            {
                IEnumerable<LichMatchingData> lichHopLe =
                    tour.LichKhoiHanhs;

                if (request.NgayDuKien.HasValue)
                {
                    DateOnly ngayYeuCau =
                        request.NgayDuKien.Value;

                    lichHopLe = lichHopLe.Where(
                        lich =>
                            lich.NgayKhoiHanh ==
                            ngayYeuCau);
                }

                if (request.SoKhach.HasValue)
                {
                    int soKhach =
                        request.SoKhach.Value;

                    lichHopLe = lichHopLe.Where(
                        lich =>
                            lich.SoChoConLai >= soKhach);
                }

                lichPhuHop = lichHopLe
                    .OrderBy(
                        lich => lich.NgayKhoiHanh)
                    .ThenBy(
                        lich => lich.IdLich)
                    .FirstOrDefault();

                if (lichPhuHop is null)
                {
                    continue;
                }
            }

            // ================================================
            // 5. TINH MATCHING SCORE
            // ================================================
            //
            // SO THICH: 50
            // DIEM DEN: 30
            // NGAN SACH: 20
            //
            // SO NGAY, LOAI TOUR, LICH, SO KHACH
            // LA DIEU KIEN LOC.
            //
            // KHONG TU THEM TRONG SO MOI.

            var lyDo =
                new List<string>();

            // 5.1. SO THICH - TOI DA 50

            int diemSoThich = 0;

            if (soThichDaChon.Count > 0)
            {
                diemSoThich = (int)Math.Round(
                    (double)DiemToiDaSoThich *
                    soThichTrung /
                    soThichDaChon.Count,
                    MidpointRounding.AwayFromZero);

                if (soThichTrung > 0)
                {
                    lyDo.Add(
                        $"Phù hợp {soThichTrung}/" +
                        $"{soThichDaChon.Count} " +
                        "sở thích đã chọn");
                }
            }

            // 5.2. DIEM DEN - TOI DA 30

            int diemDiemDen = 0;

            if (request.IdDiemDen.HasValue &&
                tour.DiemDenIds.Contains(
                    request.IdDiemDen.Value))
            {
                diemDiemDen =
                    DiemToiDaDiemDen;

                lyDo.Add(
                    "Có điểm tham quan đúng với " +
                    "điểm đến mong muốn");
            }

            // 5.3. NGAN SACH - TOI DA 20

            int diemNganSach = 0;

            if (request.NganSachToiDa.HasValue)
            {
                diemNganSach =
                    DiemToiDaNganSach;

                lyDo.Add(
                    nganSachCaNhom
                        ? "Chi phí nhóm ước tính " +
                          "nằm trong ngân sách"
                        : "Giá tour nằm trong " +
                          "ngân sách mỗi người");
            }

            // ================================================
            // 5.4. GIAI THICH CAC DIEU KIEN LOC
            // ================================================

            if (request.SoNgayToiDa.HasValue &&
                tour.SoNgay.HasValue)
            {
                lyDo.Add(
                    $"Thời gian {tour.SoNgay.Value} ngày " +
                    "đáp ứng giới hạn chuyến đi");
            }

            if (coLoaiTour)
            {
                lyDo.Add(
                    $"Thuộc loại tour {tour.TenLoaiTour}");
            }

            if (lichPhuHop is not null)
            {
                string ngay =
                    lichPhuHop.NgayKhoiHanh.ToString(
                        "dd/MM/yyyy",
                        CultureInfo.InvariantCulture);

                lyDo.Add(
                    $"Có lịch khởi hành ngày {ngay}");

                if (request.SoKhach.HasValue)
                {
                    lyDo.Add(
                        $"Lịch còn " +
                        $"{lichPhuHop.SoChoConLai} chỗ, " +
                        $"đáp ứng {request.SoKhach.Value} khách");
                }
            }

            if (lyDo.Count == 0)
            {
                lyDo.Add(
                    "Tour đang hoạt động " +
                    "và có giá bán hợp lệ");
            }

            int tongDiem =
                diemSoThich +
                diemDiemDen +
                diemNganSach;

            // ================================================
            // 6. TAO KET QUA
            // ================================================

            results.Add(
                new MatchingScoreResult
                {
                    IdTour = tour.IdTour,

                    TenTour = tour.TenTour,

                    DiaDiem = tour.DiaDiem,

                    HinhAnh = tour.HinhAnh,

                    ThoiGian = tour.ThoiGian,

                    GiaBan = tour.GiaBan,

                    SoNgay = tour.SoNgay,

                    DiemSoThich =
                        diemSoThich,

                    DiemDiemDen =
                        diemDiemDen,

                    DiemNganSach =
                        diemNganSach,

                    MatchingScore =
                        tongDiem,

                    LyDoGoiY =
                        lyDo,

                    TenLoaiTour =
                        tour.TenLoaiTour,

                    NgayKhoiHanhPhuHop =
                        lichPhuHop?.NgayKhoiHanh,

                    SoChoConLai =
                        lichPhuHop?.SoChoConLai
                });
        }

        // ====================================================
        // 7. RANKING TOP-N
        // ====================================================

        return results
            .OrderByDescending(
                tour => tour.MatchingScore)
            .ThenBy(
                tour => tour.GiaBan)
            .ThenBy(
                tour => tour.IdTour)
            .Take(topN)
            .ToList();
    }

    // ========================================================
    // HELPERS
    // ========================================================

    private static bool IsDiemDenRole(
        string? value)
    {
        return NormalizeText(value) == "diemden";
    }

    private static string NormalizeText(
        string? value)
    {
        if (string.IsNullOrWhiteSpace(value))
        {
            return "";
        }

        string normalized = value
            .Trim()
            .ToLowerInvariant()
            .Replace('đ', 'd')
            .Normalize(
                NormalizationForm.FormD);

        var builder = new StringBuilder();

        foreach (char character in normalized)
        {
            UnicodeCategory category =
                CharUnicodeInfo.GetUnicodeCategory(
                    character);

            if (category !=
                UnicodeCategory.NonSpacingMark)
            {
                builder.Append(character);
            }
        }

        return Regex.Replace(
                builder.ToString()
                    .Normalize(
                        NormalizationForm.FormC),
                @"[^a-z0-9]+",
                "")
            .Trim();
    }

    private static string ReadText(
        DbDataReader reader,
        int ordinal)
    {
        return reader.IsDBNull(ordinal)
            ? ""
            : Convert.ToString(
                reader.GetValue(ordinal),
                CultureInfo.InvariantCulture) ?? "";
    }

    private static DateOnly? ReadDateOnly(
        DbDataReader reader,
        int ordinal)
    {
        if (reader.IsDBNull(ordinal))
        {
            return null;
        }

        object value =
            reader.GetValue(ordinal);

        if (value is DateOnly dateOnly)
        {
            return dateOnly;
        }

        if (value is DateTime dateTime)
        {
            return DateOnly.FromDateTime(
                dateTime);
        }

        if (DateOnly.TryParse(
            Convert.ToString(
                value,
                CultureInfo.InvariantCulture),
            CultureInfo.InvariantCulture,
            DateTimeStyles.None,
            out DateOnly parsed))
        {
            return parsed;
        }

        return null;
    }

    // ========================================================
    // PARSE SO NGAY
    // ========================================================

    private static int? ParseSoNgay(
        string? text)
    {
        if (string.IsNullOrWhiteSpace(text))
        {
            return null;
        }

        string normalized = text
            .Trim()
            .Normalize(
                NormalizationForm.FormC);

        // Ho tro:
        // 2 ngay 1 dem
        // 2 ngày 1 đêm
        // 2N1D
        // 2N1Đ
        // 3 ngay
        // 1 ngày

        Match match = Regex.Match(
            normalized,
            @"(?<!\d)(\d+)\s*(?:ngày|ngay|n)(?![a-z])",
            RegexOptions.IgnoreCase |
            RegexOptions.CultureInvariant);

        if (!match.Success)
        {
            return null;
        }

        if (!int.TryParse(
            match.Groups[1].Value,
            NumberStyles.None,
            CultureInfo.InvariantCulture,
            out int days))
        {
            return null;
        }

        return days is >= 1 and <= 365
            ? days
            : null;
    }

    // ========================================================
    // INTERNAL DATA
    // ========================================================

    private sealed class TourMatchingData
    {
        public int IdTour { get; set; }

        public string TenTour { get; set; } = "";

        public string DiaDiem { get; set; } = "";

        public string HinhAnh { get; set; } = "";

        public string ThoiGian { get; set; } = "";

        public decimal GiaBan { get; set; }

        public int? SoNgay { get; set; }

        public int IdLoaiTour { get; set; }

        public string TenLoaiTour { get; set; } = "";

        public HashSet<int> DiemDenIds { get; } = new();

        public HashSet<int> SoThichIds { get; } = new();

        public List<LichMatchingData> LichKhoiHanhs { get; }
            = new();
    }

    private sealed class LichMatchingData
    {
        public int IdLich { get; set; }

        public DateOnly NgayKhoiHanh { get; set; }

        public int SoChoConLai { get; set; }

        public string TrangThai { get; set; } = "";
    }
}

/// <summary>
/// Keeps schedule capacity derived from active bookings. Recalculation is deliberately
/// idempotent so legacy database triggers cannot make seats drift through double updates.
/// </summary>
public sealed class SeatInventoryService : ISeatInventoryService
{
    private readonly ApplicationDbContext _db;

    public SeatInventoryService(ApplicationDbContext db) => _db = db;

    public int CountGuests(int adults, int children, int infants)
        => Math.Max(0, adults) + Math.Max(0, children) + Math.Max(0, infants);

    public async Task<int> GetRemainingSeatsAsync(int scheduleId, CancellationToken cancellationToken = default)
    {
        var capacity = await _db.LichKhoiHanhs
            .Where(x => x.IdLich == scheduleId)
            .Select(x => x.SoChoToiDa)
            .SingleOrDefaultAsync(cancellationToken);

        var reserved = await ActiveBookings(scheduleId)
            .SumAsync(x => x.NguoiLon + x.TreEm + x.TreNho, cancellationToken);

        return Math.Max(0, capacity - reserved);
    }

    public async Task SynchronizeAsync(int scheduleId, int tourId, CancellationToken cancellationToken = default)
    {
        var schedule = await _db.LichKhoiHanhs.SingleAsync(x => x.IdLich == scheduleId, cancellationToken);
        var previousRemainingSeats = schedule.SoChoConLai;
        var reservedForSchedule = await ActiveBookings(scheduleId)
            .SumAsync(x => x.NguoiLon + x.TreEm + x.TreNho, cancellationToken);

        schedule.SoChoConLai = Math.Max(0, schedule.SoChoToiDa - reservedForSchedule);
        schedule.TrangThai = schedule.SoChoConLai switch
        {
            <= 0 => "Hết chỗ",
            <= 5 => "Sắp hết chỗ",
            _ => "Còn chỗ"
        };

        var tour = await _db.Tours.SingleAsync(x => x.IdTour == tourId, cancellationToken);
        tour.SoNguoiDaDat = await _db.DonDatTours
            .Where(x => x.IdTour == tourId && x.TrangThai != BookingPaymentStatus.TrangThaiDaHuy)
            .SumAsync(x => x.NguoiLon + x.TreEm + x.TreNho, cancellationToken);

        await _db.SaveChangesAsync(cancellationToken);

        if (previousRemainingSeats <= 0 && schedule.SoChoConLai > 0)
        {
            var tourLink = $"/Tour/ChiTiet/{tourId}";
            await _db.Database.ExecuteSqlInterpolatedAsync($@"
IF OBJECT_ID(N'dbo.DanhSachCho', N'U') IS NOT NULL
BEGIN
    DECLARE @IdDanhSachCho INT;
    DECLARE @IdTaiKhoan INT;
    SELECT TOP (1) @IdDanhSachCho = IdDanhSachCho, @IdTaiKhoan = IdTaiKhoan
    FROM dbo.DanhSachCho
    WHERE IdLich = {scheduleId} AND TrangThai = N'Đang chờ' AND SoKhach <= {schedule.SoChoConLai}
    ORDER BY NgayDangKy, IdDanhSachCho;

    IF @IdDanhSachCho IS NOT NULL
    BEGIN
        UPDATE dbo.DanhSachCho SET TrangThai = N'Đã thông báo' WHERE IdDanhSachCho = @IdDanhSachCho;
        INSERT INTO dbo.ThongBao (IdDon, NoiDung, NgayTao, DaDoc, IdNguoiNhan, TieuDe, LienKet)
        VALUES (NULL, N'Lịch tour bạn chờ vừa có chỗ trống. Hãy đặt sớm trước khi hết chỗ.', GETDATE(), 0,
                @IdTaiKhoan, N'Tour đã có chỗ trở lại', {tourLink});
    END
END", cancellationToken);
        }
    }

    public async Task SynchronizeAllAsync(CancellationToken cancellationToken = default)
    {
        var reservedBySchedule = await _db.DonDatTours
            .Where(x => x.TrangThai != BookingPaymentStatus.TrangThaiDaHuy)
            .GroupBy(x => x.IdLich)
            .Select(x => new
            {
                IdLich = x.Key,
                Reserved = x.Sum(d => d.NguoiLon + d.TreEm + d.TreNho)
            })
            .ToDictionaryAsync(x => x.IdLich, x => x.Reserved, cancellationToken);

        var reservedByTour = await _db.DonDatTours
            .Where(x => x.TrangThai != BookingPaymentStatus.TrangThaiDaHuy)
            .GroupBy(x => x.IdTour)
            .Select(x => new
            {
                IdTour = x.Key,
                Reserved = x.Sum(d => d.NguoiLon + d.TreEm + d.TreNho)
            })
            .ToDictionaryAsync(x => x.IdTour, x => x.Reserved, cancellationToken);

        var schedules = await _db.LichKhoiHanhs
            .ToListAsync(cancellationToken);

        foreach (var schedule in schedules)
        {
            reservedBySchedule.TryGetValue(schedule.IdLich, out var reserved);
            schedule.SoChoConLai = Math.Max(0, schedule.SoChoToiDa - reserved);
            schedule.TrangThai = schedule.SoChoConLai switch
            {
                <= 0 => "Hết chỗ",
                <= 5 => "Sắp hết chỗ",
                _ => "Còn chỗ"
            };
        }

        var tours = await _db.Tours.ToListAsync(cancellationToken);
        foreach (var tour in tours)
        {
            reservedByTour.TryGetValue(tour.IdTour, out var reserved);
            tour.SoNguoiDaDat = reserved;
        }

        await _db.SaveChangesAsync(cancellationToken);
    }

    private IQueryable<DonDatTour> ActiveBookings(int scheduleId)
        => _db.DonDatTours.Where(x =>
            x.IdLich == scheduleId &&
            x.TrangThai != BookingPaymentStatus.TrangThaiDaHuy);
}

    // ========================================================
    // 1. TIM TOUR
    // ========================================================

    public async Task<TripiChatMatchingResult> TimTourAsync(
        TripiChatRequest request,
        CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(request);

        if (string.IsNullOrWhiteSpace(request.Message))
        {
            throw new ArgumentException(
                "Nội dung yêu cầu không được để trống.");
        }

        var danhSachDiemDen = new List<TripiDanhMucItem>();
        var danhSachSoThich = new List<TripiDanhMucItem>();
        var danhSachLoaiTour = new List<TripiDanhMucItem>();

        var connection = _db.Database.GetDbConnection();

        bool openedHere =
            connection.State != ConnectionState.Open;

        if (openedHere)
        {
            await connection.OpenAsync(cancellationToken);
        }

        try
        {
            // ================================================
            // 2. DANH MUC DIEM DEN
            // ================================================

            await using (var command = connection.CreateCommand())
            {
                command.CommandText = """
                    SELECT
                        IdDiemDen,
                        TenDiemDen
                    FROM dbo.AIDiemDen
                    WHERE TrangThai = 1
                    ORDER BY TenDiemDen;
                    """;

                await using var reader =
                    await command.ExecuteReaderAsync(
                        cancellationToken);

                while (await reader.ReadAsync(cancellationToken))
                {
                    danhSachDiemDen.Add(
                        new TripiDanhMucItem
                        {
                            Id = reader.GetInt32(0),
                            Ten = reader.GetString(1)
                        });
                }
            }

            // ================================================
            // 3. DANH MUC SO THICH
            // ================================================

            await using (var command = connection.CreateCommand())
            {
                command.CommandText = """
                    SELECT
                        IdSoThich,
                        TenSoThich
                    FROM dbo.AISoThich
                    WHERE TrangThai = 1
                    ORDER BY TenSoThich;
                    """;

                await using var reader =
                    await command.ExecuteReaderAsync(
                        cancellationToken);

                while (await reader.ReadAsync(cancellationToken))
                {
                    danhSachSoThich.Add(
                        new TripiDanhMucItem
                        {
                            Id = reader.GetInt32(0),
                            Ten = reader.GetString(1)
                        });
                }
            }

            // ================================================
            // 4. DANH MUC LOAI TOUR
            // ================================================

            await using (var command = connection.CreateCommand())
            {
                command.CommandText = """
                    SELECT
                        lt.IdLoaiTour,
                        lt.TenLoai
                    FROM dbo.LoaiTour AS lt
                    WHERE EXISTS
                    (
                        SELECT 1
                        FROM dbo.Tour AS t
                        WHERE t.IdLoaiTour = lt.IdLoaiTour
                          AND t.TrangThai = 1
                    )
                    ORDER BY lt.TenLoai;
                    """;

                await using var reader =
                    await command.ExecuteReaderAsync(
                        cancellationToken);

                while (await reader.ReadAsync(cancellationToken))
                {
                    danhSachLoaiTour.Add(
                        new TripiDanhMucItem
                        {
                            Id = reader.GetInt32(0),
                            Ten = reader.GetString(1)
                        });
                }
            }
        }
        finally
        {
            if (openedHere)
            {
                await connection.CloseAsync();
            }
        }

        // ====================================================
        // 5. KIEM TRA DANH MUC
        // ====================================================

        if (danhSachDiemDen.Count == 0 &&
            danhSachSoThich.Count == 0 &&
            danhSachLoaiTour.Count == 0)
        {
            return new TripiChatMatchingResult
            {
                ThongBao =
                    "Hệ thống chưa có danh mục phù hợp " +
                    "để phân tích nhu cầu du lịch."
            };
        }

        // ====================================================
        // 6. CHUYEN HOI THOAI SANG GEMINI
        // ====================================================

        string noiDungHoiThoai =
            TaoNoiDungHoiThoai(request);

        var geminiRequest = new GeminiTravelRequest
        {
            NoiDung = noiDungHoiThoai,

            DiemDens = danhSachDiemDen
                .Select(x => x.Ten)
                .Take(60)
                .ToList(),

            SoThichs = danhSachSoThich
                .Select(x => x.Ten)
                .Take(60)
                .ToList(),

            LoaiTours = danhSachLoaiTour
                .Select(x => x.Ten)
                .Take(60)
                .ToList()
        };

        var geminiService =
            new GeminiService(_configuration);

        GeminiTravelResult phanTich =
            await geminiService.PhanTichAsync(
                geminiRequest,
                cancellationToken);

        // ====================================================
        // 7. DOI CHIEU DU LIEU VOI DANH MUC SQL
        // ====================================================

        var ketQua = new TripiChatMatchingResult();

        TripiDanhMucItem? diemDenHopLe = null;

        if (!string.IsNullOrWhiteSpace(phanTich.DiemDen))
        {
            diemDenHopLe = danhSachDiemDen.FirstOrDefault(
                x => string.Equals(
                    x.Ten.Trim(),
                    phanTich.DiemDen.Trim(),
                    StringComparison.OrdinalIgnoreCase));
        }

        var soThichHopLe = danhSachSoThich
            .Where(x => (phanTich.SoThich ??
                         new List<string>())
                .Any(ten => string.Equals(
                    x.Ten.Trim(),
                    ten.Trim(),
                    StringComparison.OrdinalIgnoreCase)))
            .GroupBy(x => x.Id)
            .Select(group => group.First())
            .ToList();

        TripiDanhMucItem? loaiTourHopLe = null;

        if (!string.IsNullOrWhiteSpace(phanTich.LoaiTour))
        {
            loaiTourHopLe = danhSachLoaiTour.FirstOrDefault(
                x => string.Equals(
                    x.Ten.Trim(),
                    phanTich.LoaiTour.Trim(),
                    StringComparison.OrdinalIgnoreCase));
        }

        // ====================================================
        // 8. LUU CAC TIEU CHI DA NHAN DIEN
        // ====================================================

        ketQua.DiemDen = diemDenHopLe?.Ten;

        ketQua.SoThich = soThichHopLe
            .Select(x => x.Ten)
            .ToList();

        ketQua.NganSach = phanTich.NganSach;

        ketQua.LoaiNganSach =
            ChuanHoaLoaiNganSach(phanTich.LoaiNganSach);
        
        // Xac minh pham vi ngan sach bang loi khach.
        // Khong tin tuyet doi pham vi do Gemini suy doan.

        var tinNhanKhach = (request.History ??
            new List<TripiChatMessage>())
            .Where(x => x.Role == "user")
            .Select(x => x.Content)
            .Append(request.Message);

        bool coBangChungPhamVi = tinNhanKhach.Any(text =>
            !string.IsNullOrWhiteSpace(text) &&
            System.Text.RegularExpressions.Regex.IsMatch(
                text,
                @"mỗi\s*(người|đứa|khách|vé)" +
                @"|một\s*người|/người|/khách" +
                @"|tổng\s*(ngân sách|cộng|chi phí|tiền)" +
                @"|(?:ngân sách|chi phí|tiền)\s*(?:cho|của)\s*cả\s*(?:nhóm|đoàn|nhà)" +
                @"|cả\s*(?:nhóm|đoàn|nhà)\s*(?:chỉ\s*)?(?:có|góp|dành)\s*(?:tổng\s*cộng\s*)?\d+",
                System.Text.RegularExpressions.RegexOptions.IgnoreCase |
                System.Text.RegularExpressions.RegexOptions.CultureInvariant));

        
        // Chi ke thua pham vi ngan sach khi khach
        // dang dieu chinh nhu cau da co.
        bool dangDieuChinh = 
            System.Text.RegularExpressions.Regex.IsMatch(
                request.Message,
                @"giữ nguyên|giu nguyen|đổi|doi|thay|tăng|tang|giảm|giam|chuyển|chuyen|ngân sách mới|ngan sach moi",
                System.Text.RegularExpressions.RegexOptions.IgnoreCase);

        bool coPhamViDaLuu =
            request.Preferences?.LoaiNganSach is
                "MoiNguoi" or "CaNhom";

        if (ketQua.NganSach.HasValue &&
            !coBangChungPhamVi)
        {
            if (dangDieuChinh && coPhamViDaLuu)
            {
                ketQua.LoaiNganSach =
                    request.Preferences!.LoaiNganSach;
            }
            else
            {
                ketQua.LoaiNganSach = null;
            }
        }
        
        ketQua.SoNgay = phanTich.SoNgay;

        // Khi khach yeu cau giu nguyen cac dieu kien,
        // khong lam mat so ngay cu neu Gemini bo sot.
        string tinNhanMoi = request.Message ?? "";

        bool yeuCauGiuNguyen =
            System.Text.RegularExpressions.Regex.IsMatch(
                tinNhanMoi,
                @"giữ nguyên|giu nguyen|như cũ|nhu cu",
                System.Text.RegularExpressions.RegexOptions.IgnoreCase);

        bool coYeuCauSuaSoNgay =
            System.Text.RegularExpressions.Regex.IsMatch(
                tinNhanMoi,
                @"\d+\s*(ngày|ngay)\b" +
                @"|(?:bỏ|bo|xóa|xoa|không giới hạn|khong gioi han|không cần|khong can|thay|đổi|doi|tăng|tang|giảm|giam)" +
                @".{0,40}(?:số ngày|so ngay|ngày|ngay|thời gian|thoi gian)",
                System.Text.RegularExpressions.RegexOptions.IgnoreCase);

        if (!ketQua.SoNgay.HasValue &&
            yeuCauGiuNguyen &&
            !coYeuCauSuaSoNgay &&
            request.Preferences?.SoNgay is int soNgayCu &&
            soNgayCu > 0)
        {
            ketQua.SoNgay = soNgayCu;
        }

        ketQua.LoaiTour = loaiTourHopLe?.Ten;

        ketQua.SoKhach = phanTich.SoKhach;

        ketQua.NgayDuKien = phanTich.NgayDuKien;

        // Khong luu pham vi ngan sach khi chua co so tien.
        if (!ketQua.NganSach.HasValue)
        {
            ketQua.LoaiNganSach = null;
        }

        ketQua.CoTieuChi =
            diemDenHopLe != null ||
            soThichHopLe.Count > 0 ||
            ketQua.NganSach.HasValue ||
            ketQua.SoNgay.HasValue ||
            loaiTourHopLe != null ||
            ketQua.SoKhach.HasValue ||
            ketQua.NgayDuKien.HasValue;

        bool coTieuChiTimKiem =
            diemDenHopLe != null ||
            soThichHopLe.Count > 0 ||
            ketQua.NganSach.HasValue ||
            loaiTourHopLe != null ||
            ketQua.SoKhach.HasValue ||
            ketQua.NgayDuKien.HasValue;

        if (!coTieuChiTimKiem)
        {
            ketQua.ThongBao = ketQua.CoTieuChi
                ? "Tripi đã nhận diện thời lượng chuyến đi. " +
                  "Bạn muốn đến đâu, thích trải nghiệm gì " +
                  "hoặc có ngân sách bao nhiêu?"
                : "Tripi chưa xác định được nhu cầu tìm tour. " +
                  "Bạn hãy cho biết điểm đến, sở thích, " +
                  "loại tour hoặc ngân sách nhé.";

            return ketQua;
        }

        // ====================================================
        // 9. KIEM TRA PHAM VI NGAN SACH
        // ====================================================

        // Neu khach da cung cap ngan sach,
        // phai xac dinh ngan sach la moi nguoi
        // hay cho ca nhom.
        //
        // Khong ngam hieu null thanh MoiNguoi.

        if (ketQua.NganSach.HasValue &&
            ketQua.LoaiNganSach == null)
        {
            ketQua.ThongBao =
                "Mình đã ghi nhận ngân sách " +
                DinhDangTien(ketQua.NganSach.Value) +
                ". Đây là ngân sách mỗi người hay " +
                "tổng ngân sách cho cả nhóm vậy bạn? " +
                "Mình cần biết để lọc tour chính xác hơn.";

            return ketQua;
        }

        // Neu ngan sach ca nhom thi phai biet
        // so khach moi co the tinh chi phi uoc tinh.

        if (ketQua.NganSach.HasValue &&
            ketQua.LoaiNganSach == "CaNhom" &&
            !ketQua.SoKhach.HasValue)
        {
            ketQua.ThongBao =
                "Mình đã ghi nhận tổng ngân sách " +
                DinhDangTien(ketQua.NganSach.Value) +
                " cho cả nhóm. Bạn cho mình biết " +
                "có bao nhiêu người tham gia nhé, " +
                "để Tripi tính chi phí phù hợp.";

            return ketQua;
        }

        // ====================================================
        // 10. CHUYEN DU LIEU SANG MATCHING SCORE
        // ====================================================

        var matchingRequest = new MatchingScoreRequest
        {
            IdDiemDen = diemDenHopLe?.Id,

            IdSoThich = soThichHopLe
                .Select(x => x.Id)
                .Distinct()
                .ToList(),

            NganSachToiDa = ketQua.NganSach,

            // FIX: Truyen pham vi ngan sach sang Matching.
            LoaiNganSach = ketQua.LoaiNganSach,

            SoNgayToiDa = ketQua.SoNgay,

            LoaiTour = ketQua.LoaiTour,

            SoKhach = ketQua.SoKhach,

            NgayDuKien = ketQua.NgayDuKien,

            TopN = 5
        };

        ketQua.Tours =
            await _matchingService.GoiYTourAsync(
                matchingRequest,
                cancellationToken);

        ketQua.DaTimKiem = true;

        // ====================================================
        // 11. THONG BAO KET QUA
        // ====================================================

        if (ketQua.Tours.Count > 0)
        {
            ketQua.ThongBao =
                $"Tripi đã tìm thấy {ketQua.Tours.Count} " +
                "tour theo những tiêu chí đã xác định. " +
                "Các tour được xếp hạng bằng Matching Score.";

            if (ketQua.LoaiNganSach == "CaNhom")
            {
                ketQua.ThongBao +=
                    " Chi phí nhóm hiện được ước tính " +
                    "theo giá tour cơ bản nhân số khách; " +
                    "tổng tiền đặt tour thực tế có thể " +
                    "khác nếu có giá trẻ em hoặc ưu đãi.";
            }
        }
        else
        {
            if (ketQua.NgayDuKien.HasValue)
            {
                ketQua.ThongBao =
                    "Hiện chưa tìm thấy tour có lịch " +
                    "khởi hành đúng ngày " +
                    ketQua.NgayDuKien.Value.ToString(
                        "dd/MM/yyyy") +
                    " và đáp ứng các điều kiện còn lại. " +
                    "Bạn có thể thử một ngày khác " +
                    "hoặc thay đổi số khách.";
            }
            else
            {
                ketQua.ThongBao =
                    "Hiện chưa tìm thấy tour đáp ứng " +
                    "đầy đủ các tiêu chí bạn yêu cầu. " +
                    "Bạn có thể thử điều chỉnh điểm đến, " +
                    "sở thích, thời lượng hoặc ngân sách.";
            }
        }

        return ketQua;
    }

    private static string? ChuanHoaLoaiNganSach(
        string? value)
    {
        if (string.Equals(
            value,
            "MoiNguoi",
            StringComparison.OrdinalIgnoreCase))
        {
            return "MoiNguoi";
        }

        if (string.Equals(
            value,
            "CaNhom",
            StringComparison.OrdinalIgnoreCase))
        {
            return "CaNhom";
        }

        return null;
    }

                "hoặc xóa bất kỳ tiêu chí nào.");

            builder.AppendLine(
                "Nếu tin nhắn mới không thay đổi tiêu chí nào, " +
                "giữ giá trị cũ đã được xác định.");

            builder.AppendLine(
                "Không được tự thêm tiêu chí mới. " +
                "Không tự đoán phạm vi ngân sách.");

            builder.AppendLine();
        }

        builder.AppendLine("LỊCH SỬ HỘI THOẠI:");

        var history = (request.History ??
                       new List<TripiChatMessage>())
            .Where(x =>
                x != null &&
                x.Role == "user" &&
                !string.IsNullOrWhiteSpace(x.Content))
            .TakeLast(12)
            .ToList();

        foreach (var item in history)
        {
            string content = item.Content.Trim();

            if (content.Length > 1000)
            {
                content = content[..1000];
            }

            builder.AppendLine($"Khách: {content}");
        }

        builder.AppendLine();

        builder.AppendLine("TIN NHẮN MỚI NHẤT:");

        string currentMessage = request.Message.Trim();

        if (currentMessage.Length > 1000)
        {
            currentMessage = currentMessage[..1000];
        }

        builder.AppendLine($"Khách: {currentMessage}");

        return builder.ToString();
    }

        // Không bắt khách phải nói chính xác "tìm tour".

                        Regex.IsMatch(

                            @"\b\d+\s*(ngay|dem|nguoi|khach)\b|\b\d+(?:[.,]\d+)?\s*(trieu|tr|nghin|k)\b")
                    )
                ));

        // Khong phu thuoc hoan toan vao 12 tin nhan gan nhat.
        bool hasStoredSearch =
            request.Preferences is { } saved &&
            (
                !string.IsNullOrWhiteSpace(saved.DiemDen) ||
                (saved.SoThich?.Count ?? 0) > 0 ||
                saved.NganSach.HasValue ||
                saved.SoNgay.HasValue ||
                !string.IsNullOrWhiteSpace(saved.LoaiTour) ||
                saved.SoKhach.HasValue ||
                !string.IsNullOrWhiteSpace(saved.NgayDuKien)
            );

        previousSearch = previousSearch || hasStoredSearch;

        bool explicitRefinement =
            HasExplicitRefinement(userMessage);

        // Ưu tiên yêu cầu tìm tour rõ ràng.
        // Không phủ nhận kết quả Gemini nếu Gemini
        // đã phân loại search_tour.
        if (explicitSearch || naturalSearch)
        {
            intent = "search_tour";
        }

        // KHACH CHUA BIET DI DAU - TU VAN TRUOC KHI TIM TOUR
        // ====================================================

        bool dangPhanVanDiemDen =
            Regex.IsMatch(
                normalizedMessage,
                @"\b(chua biet|khong biet|phan van|chua quyet dinh)\b")
            &&
            Regex.IsMatch(
                normalizedMessage,
                @"\b(di dau|noi nao|dia diem|diem den|di du lich|di choi|di bien|di nui)\b");

        // Khach chua biet di dau thi khong tu dong
        // chuyen sang Matching Score, tru khi khach
        // da chu dong yeu cau tim tour ro rang.
        if (dangPhanVanDiemDen &&
            !explicitSearch &&
            !naturalSearch &&
            !explicitRefinement)
        {
            intent = "general";

            // Neu Gemini da soan cau tra loi theo huong
            // yeu cau tim tour, thay bang loi tu van.
            if (string.IsNullOrWhiteSpace(reply) ||
                Regex.IsMatch(
                    Normalize(reply),
                    @"chua xac dinh duoc nhu cau|hay cho biet diem den|can them tieu chi"))
            {
                reply =
                    "Được chứ! 🦊 Nếu bạn chưa biết nên đi đâu, " +
                    "mình sẽ cùng bạn tìm một điểm đến phù hợp nhé.\n\n" +
                    "Bạn thích đi biển để nghỉ dưỡng, " +
                    "khám phá thiên nhiên hay tham quan " +
                    "những địa điểm văn hóa, lịch sử?\n\n" +
                    "Mình có thể dựa vào sở thích của bạn " +
                    "để tư vấn trước, chưa cần chọn tour ngay đâu!";
            }
        }

        // Khi khách thay đổi điều kiện của chuyến đi
        // đã yêu cầu tìm trước đó, tiếp tục tìm lại.
        // Không tự tìm khi chỉ đang trò chuyện.
        bool canSearch =
            intent == "search_tour" ||
            (
                intent == "refine" &&
                previousSearch &&
                explicitRefinement
            );

        return new TripiChatResponse
        {
            Reply = reply,
            Intent = intent,
            CanSearchTours = canSearch
        };
    }

    {

            "goi y hanh trinh",
            "de xuat tour",
            "tu van tour",
            "chon tour",
            "xem tour",
            "danh sach tour",
            "co tour nao",
            "tour nao phu hop",
            "cho xem tour",
            "muon dat tour",
            "dat tour di",
            "tim lai tour",
            "loc tour",
            "tour phu hop voi minh",
            "tour phu hop cho minh"
        };

        if (phrases.Any(phrase =>

        bool mentionsTour =

        bool asksForSuggestions =
            Regex.IsMatch(
                normalized,
                @"\b(tim|goi y|de xuat|tu van|liet ke)\b");

        return mentionsTour && asksForSuggestions;
    }

        };

        if (phrases.Any(phrase =>

        {

        }

        bool expressesChange =
            Regex.IsMatch(

        bool mentionsAmount =
            Regex.IsMatch(

        if (mentionsAmount && expressesChange)
        {
            return true;
        }

        bool mentionsDays =
            Regex.IsMatch(
                normalized,
                @"\b\d+\s*ngay\b");

        if (mentionsDays && expressesChange)
        {
            return true;
        }

        bool mentionsGuests =
            Regex.IsMatch(

        if (mentionsSpecificDate &&
            Regex.IsMatch(
                normalized,

        {
            return true;
        }

        return false;
    }

    // ========================================================
    // NORMALIZE VIETNAMESE
    // ========================================================

    private static string Normalize(string text)
    {
        if (string.IsNullOrWhiteSpace(text))
        {
            return "";
        }

        string decomposed = text
            .Trim()
            .ToLowerInvariant()
            .Replace('đ', 'd')
            .Normalize(NormalizationForm.FormD);

        var builder = new StringBuilder();

        foreach (char character in decomposed)
        {
            UnicodeCategory category =
                CharUnicodeInfo.GetUnicodeCategory(
                    character);

            if (category != UnicodeCategory.NonSpacingMark)
            {
                builder.Append(character);
            }
        }

        string normalized =
            builder.ToString()
                .Normalize(NormalizationForm.FormC);

        normalized = Regex.Replace(
            normalized,
            @"[^a-z0-9]+",
            " ");

        return Regex.Replace(
            normalized,
            @"\s+",
            " ").Trim();
    }

    // EXTRACT GEMINI RESPONSE

    private static string ExtractResponseText(
        JsonDocument document)
    {
        JsonElement root = document.RootElement;

        if (!root.TryGetProperty(
                "candidates",
                out JsonElement candidates) ||
            candidates.ValueKind != JsonValueKind.Array ||
            candidates.GetArrayLength() == 0)
        {
            throw new InvalidOperationException(
                "Gemini chưa trả về câu trả lời.");
        }

        JsonElement candidate = candidates[0];

        if (!candidate.TryGetProperty(
                "content",
                out JsonElement content) ||
            !content.TryGetProperty(
                "parts",
                out JsonElement parts) ||
            parts.ValueKind != JsonValueKind.Array)
        {
            throw new InvalidOperationException(
                "Gemini trả về nội dung chưa hợp lệ.");
        }

        var texts = new List<string>();

        foreach (JsonElement part in parts.EnumerateArray())
        {
            if (part.TryGetProperty(
                    "text",
                    out JsonElement value) &&
                value.ValueKind == JsonValueKind.String)
            {
                string? text = value.GetString();

                if (!string.IsNullOrWhiteSpace(text))
                {
                    texts.Add(text);
                }
            }
        }

        if (texts.Count == 0)
        {
            throw new InvalidOperationException(
                "Gemini chưa tạo được câu trả lời.");
        }

        return string.Join("\n", texts).Trim();
    }

    private static string ReadString(
        JsonElement root,
        string propertyName)
    {
        if (root.ValueKind != JsonValueKind.Object ||
            !root.TryGetProperty(
                propertyName,
                out JsonElement value) ||
            value.ValueKind != JsonValueKind.String)
        {
            return "";
        }

        return value.GetString()?.Trim() ?? "";
    }

    private static async Task<JsonDocument>
        SendGeminiRequestAsync(
            string primaryModel,
            string fallbackModel,
            string apiKey,
            object payload,
            CancellationToken cancellationToken)
    {
        try
        {
            // Bước 1: Thử model chính.
            return await SendToModelAsync(
                primaryModel,
                apiKey,
                payload,
                MaxAttempts,
                cancellationToken);
        }
        catch (GeminiHttpException ex) when (
            !string.IsNullOrWhiteSpace(fallbackModel) &&
            ex.StatusCode is
                HttpStatusCode.ServiceUnavailable or
                HttpStatusCode.BadGateway or
                HttpStatusCode.GatewayTimeout or
                HttpStatusCode.InternalServerError)
        {
            // Bước 2: Chỉ dùng model dự phòng khi
            // model chính gặp lỗi dịch vụ.
            //
            // Không tự chuyển model với lỗi:
            // 400, 401, 403, 404 hoặc 429.

            try
            {
                return await SendToModelAsync(
                    fallbackModel,
                    apiKey,
                    payload,
                    FallbackAttempts,
                    cancellationToken);
            }
            catch (GeminiHttpException fallbackError)
            {
                throw new InvalidOperationException(
                    $"Gemini chính ({primaryModel}) lỗi HTTP " +
                    $"{(int)ex.StatusCode}; " +
                    $"model dự phòng ({fallbackModel}) lỗi HTTP " +
                    $"{(int)fallbackError.StatusCode}: " +
                    fallbackError.Message,
                    fallbackError);
            }
        }
        catch (GeminiHttpException ex)
        {
            throw new InvalidOperationException(
                $"Gemini API lỗi HTTP " +
                $"{(int)ex.StatusCode}: " +
                ex.Message,
                ex);
        }
    }

    // ========================================================
    // SEND WITH RETRY
    // ========================================================

    private static async Task<JsonDocument>
        SendToModelAsync(
            string model,
            string apiKey,
            object payload,
            int maxAttempts,
            CancellationToken cancellationToken)
    {
        string endpoint = BuildEndpoint(model);

        for (int attempt = 1;
             attempt <= maxAttempts;
             attempt++)
        {
            cancellationToken.ThrowIfCancellationRequested();

            using var request =
                new HttpRequestMessage(
                    HttpMethod.Post,
                    endpoint);

            request.Headers.Add(
                "x-goog-api-key",
                apiKey);

            request.Content =
                JsonContent.Create(payload);

            HttpResponseMessage response;

            try
            {
                response = await Client.SendAsync(
                    request,
                    cancellationToken);
            }
            catch (HttpRequestException)
                when (attempt < maxAttempts)
            {
                await Task.Delay(
                    GetRetryDelay(attempt, null),
                    cancellationToken);

                continue;
            }

            using (response)
            {
                if (response.IsSuccessStatusCode)
                {
                    await using Stream stream =
                        await response.Content.ReadAsStreamAsync(
                            cancellationToken);

                    return await JsonDocument.ParseAsync(
                        stream,
                        cancellationToken: cancellationToken);
                }

                bool retryable =
                    response.StatusCode is
                        HttpStatusCode.TooManyRequests or
                        HttpStatusCode.InternalServerError or
                        HttpStatusCode.BadGateway or
                        HttpStatusCode.ServiceUnavailable or
                        HttpStatusCode.GatewayTimeout;

                if (retryable && attempt < maxAttempts)
                {
                    await Task.Delay(
                        GetRetryDelay(attempt, response),
                        cancellationToken);

                    continue;
                }

                string errorBody =
                    await response.Content.ReadAsStringAsync(
                        cancellationToken);

                throw new GeminiHttpException(
                    response.StatusCode,
                    GetGoogleErrorMessage(errorBody));
            }
        }

        throw new InvalidOperationException(
            "Không thể kết nối Gemini sau các lần thử.");
    }

    // ========================================================
    // EXPONENTIAL BACKOFF + JITTER
    // ========================================================

    private static TimeSpan GetRetryDelay(
        int attempt,
        HttpResponseMessage? response)
    {
        double seconds = Math.Min(
            Math.Pow(2, attempt - 1),
            8);

        var retryAfter =
            response?.Headers.RetryAfter;

        if (retryAfter?.Delta is TimeSpan delta &&
            delta > TimeSpan.Zero)
        {
            seconds = Math.Clamp(
                delta.TotalSeconds,
                1,
                12);
        }
        else if (
            retryAfter?.Date is DateTimeOffset date &&
            date > DateTimeOffset.UtcNow)
        {
            seconds = Math.Clamp(
                (date - DateTimeOffset.UtcNow)
                    .TotalSeconds,
                1,
                12);
        }

        // Jitter: tránh các request thử lại
        // cùng một thời điểm khi API quá tải.

        return TimeSpan.FromSeconds(
            seconds +
            Random.Shared.Next(100, 501) / 1000.0);
    }

    // ========================================================
    // PARSE GOOGLE ERROR
    // ========================================================

    private static string GetGoogleErrorMessage(
        string errorBody)
    {
        try
        {
            using JsonDocument document =
                JsonDocument.Parse(errorBody);

            if (document.RootElement.TryGetProperty(
                    "error",
                    out JsonElement error) &&
                error.TryGetProperty(
                    "message",
                    out JsonElement message) &&
                message.ValueKind == JsonValueKind.String)
            {
                string? value =
                    message.GetString();

                if (!string.IsNullOrWhiteSpace(value))
                {
                    return value;
                }
            }
        }
        catch (JsonException)
        {
            // Google có thể trả lỗi không phải JSON.
        }

        return "Không có thông tin chi tiết từ Google.";
    }
}

    public async Task<TripiRagResult> TraLoiVeTourAsync(
        int idTour,
        string cauHoi,
        IReadOnlyList<TripiChatMessage>? history,
        CancellationToken cancellationToken = default)
    {
        if (idTour <= 0)
        {
            throw new ArgumentException(
                "Mã tour không hợp lệ.");
        }

        if (string.IsNullOrWhiteSpace(cauHoi))
        {
            throw new ArgumentException(
                "Bạn hãy nhập câu hỏi về tour nhé.");
        }

        cauHoi = cauHoi.Trim();

        if (cauHoi.Length > MaxQuestionLength)
        {
            throw new ArgumentException(
                "Câu hỏi không được vượt quá 1000 ký tự.");
        }

        // 1. Lấy tài liệu tour đã được duyệt.

        List<TripiRagDocument> documents =
            await LayTaiLieuDaDuyetAsync(
                idTour,
                cancellationToken);

        if (documents.Count == 0)
        {
            return KhongCoBangChung(
                idTour,
                "Mình chưa có tài liệu đã được duyệt " +
                "để xác nhận thông tin của tour này. " +
                "Bạn có thể xem trang chi tiết tour " +
                "hoặc hỏi mình nội dung khác nhé.");
        }

        // 2. Xây dựng ngữ cảnh có gắn mã nguồn.

        var selectedDocuments =
            new List<TripiRagDocument>();

        string context = TaoNguCanh(
            documents,
            cauHoi,
            selectedDocuments);

        if (selectedDocuments.Count == 0 ||
            string.IsNullOrWhiteSpace(context))
        {
            return KhongCoBangChung(
                idTour,
                "Mình chưa tìm được nội dung phù hợp " +
                "để trả lời câu hỏi này từ tài liệu tour. " +
                "Mình không muốn đoán sai thông tin cho bạn.");
        }

        // 3. Gemini trả lời và chỉ ra nguồn sử dụng.

        RagGeminiAnswer answer =
            await GoiGeminiAsync(
                cauHoi,
                context,
                history,
                cancellationToken);

        // 4. Chỉ giữ mã tài liệu thực sự nằm
        // trong ngữ cảnh đã gửi cho Gemini.

        HashSet<long> citedIds =
            answer.CitedSourceIds.ToHashSet();

        List<TripiRagSource> citedSources =
            selectedDocuments
                .Where(d => citedIds.Contains(d.IdTaiLieu))
                .GroupBy(d => d.IdTaiLieu)
                .Select(g => g.First())
                .Select(d => new TripiRagSource
                {
                    IdTaiLieu = d.IdTaiLieu,
                    IdTour = d.IdTour,
                    TieuDe = d.TieuDe,
                    PhienBan = d.PhienBan
                })
                .ToList();

        // Có tài liệu được truy xuất không đồng nghĩa
        // câu trả lời đã có bằng chứng.
        //
        // Đây là kiểm tra nguồn tham chiếu cơ bản;
        // không phải xác minh độc lập từng phát biểu.

        bool hasEvidence =
            answer.Supported &&
            citedSources.Count > 0;

        if (!hasEvidence)
        {
            return KhongCoBangChung(
                idTour,
                "Mình chưa tìm thấy thông tin đủ rõ " +
                "trong tài liệu HappyTrip để xác nhận " +
                "điều bạn hỏi. " +
                "Bạn có thể hỏi về các địa điểm, " +
                "hoạt động hoặc lịch trình đã được " +
                "công bố của tour nhé.");
        }

        return new TripiRagResult
        {
            HasEvidence = true,
            IdTour = idTour,
            Reply = answer.Reply,
            Sources = citedSources
        };
    }

    private async Task<List<TripiRagDocument>>
        LayTaiLieuDaDuyetAsync(
            int idTour,
            CancellationToken cancellationToken)
    {
        var documents = new List<TripiRagDocument>();

        DbConnection connection =
            _db.Database.GetDbConnection();

        bool openedHere =
            connection.State != ConnectionState.Open;

        if (openedHere)
        {
            await connection.OpenAsync(cancellationToken);
        }

        try
        {
            await using DbCommand command =
                connection.CreateCommand();

            command.CommandText = """
                SELECT TOP (5)
                    IdTaiLieu,
                    IdTour,
                    TieuDe,
                    NoiDungGoc,
                    PhienBan
                FROM dbo.AITaiLieuRAG
                WHERE LoaiNguon = N'Tour'
                  AND IdTour = @IdTour
                  AND TrangThai = N'DaDuyet'
                  AND NoiDungGoc IS NOT NULL
                  AND LEN(NoiDungGoc) > 0
                ORDER BY
                    PhienBan DESC,
                    NgayCapNhat DESC,
                    IdTaiLieu DESC;
                """;

            DbParameter parameter =
                command.CreateParameter();

            parameter.ParameterName = "@IdTour";
            parameter.DbType = DbType.Int32;
            parameter.Value = idTour;

            command.Parameters.Add(parameter);

            await using DbDataReader reader =
                await command.ExecuteReaderAsync(
                    cancellationToken);

            while (await reader.ReadAsync(
                cancellationToken))
            {
                documents.Add(new TripiRagDocument
                {
                    IdTaiLieu = reader.GetInt64(0),
                    IdTour = reader.GetInt32(1),
                    TieuDe = reader.IsDBNull(2)
                        ? ""
                        : reader.GetString(2),
                    NoiDungGoc = reader.GetString(3),
                    PhienBan = reader.IsDBNull(4)
                        ? 1
                        : reader.GetInt32(4)
                });
            }
        }
        finally
        {
            if (openedHere)
            {
                await connection.CloseAsync();
            }
        }

        return documents
            .Take(MaxSourceDocuments)
            .ToList();
    }

    // ========================================================
    // TẠO NGỮ CẢNH CHO GEMINI
    // ========================================================

    private static string TaoNguCanh(
        List<TripiRagDocument> documents,
        string question,
        List<TripiRagDocument> selectedDocuments)
    {
        var builder = new StringBuilder();

        foreach (TripiRagDocument document in documents)
        {
            string content =
                document.NoiDungGoc.Trim();

            if (string.IsNullOrWhiteSpace(content))
            {
                continue;
            }

            string header =
                $"[TAI_LIEU_ID: {document.IdTaiLieu}]\n" +
                $"[TOUR_ID: {document.IdTour}]\n" +
                $"[PHIEN_BAN: {document.PhienBan}]\n" +
                $"TIEU_DE: {document.TieuDe}\n" +
                "NOI_DUNG:\n";

            int remaining =
                MaxContextCharacters -
                builder.Length -
                header.Length -
                3;

            if (remaining < 100)
            {
                break;
            }

            string selectedContent =
                content.Length <= remaining
                    ? content
                    : ChonDoanLienQuan(
                        content,
                        question,
                        remaining);

            if (string.IsNullOrWhiteSpace(
                selectedContent))
            {
                continue;
            }

            builder.AppendLine(header);
            builder.AppendLine(selectedContent);
            builder.AppendLine();

            selectedDocuments.Add(document);
        }

        return builder.ToString();
    }

    // ========================================================
    // CHỌN ĐOẠN LIÊN QUAN TRONG TÀI LIỆU DÀI
    // ========================================================

    private static string ChonDoanLienQuan(
        string content,
        string question,
        int maxLength)
    {
        if (maxLength <= 0)
        {
            return "";
        }

        if (content.Length <= maxLength)
        {
            return content;
        }

        HashSet<string> keywords =
            LayTuKhoa(question);

        List<string> paragraphs =
            Regex.Split(
                content,
                @"(?:\r?\n){1,}")
                .Select(p => p.Trim())
                .Where(p => p.Length > 0)
                .ToList();

        if (paragraphs.Count == 0)
        {
            return content[
                ..Math.Min(content.Length, maxLength)];
        }

        var ranked = paragraphs
            .Select((paragraph, index) => new
            {
                Text = paragraph,
                Index = index,
                Score = keywords.Count(keyword =>
                    Normalize(paragraph).Contains(
                        keyword,
                        StringComparison.Ordinal))
            })
            .OrderByDescending(x => x.Score)
            .ThenBy(x => x.Index)
            .ToList();

        var chosen =
            new List<(int Index, string Text)>();

        int used = 0;

        foreach (var paragraph in ranked)
        {
            int available = maxLength - used - 2;

            if (available <= 0)
            {
                break;
            }

            // Nếu một đoạn quá dài, chỉ lấy phần
            // phù hợp trong giới hạn còn lại.

            string selectedText = paragraph.Text;

            if (selectedText.Length > available)
            {
                if (chosen.Count > 0)
                {
                    continue;
                }

                selectedText = selectedText[..available];
            }

            if (string.IsNullOrWhiteSpace(selectedText))
            {
                continue;
            }

            chosen.Add((
                paragraph.Index,
                selectedText));

            used += selectedText.Length + 2;
        }

        if (chosen.Count == 0)
        {
            return content[
                ..Math.Min(content.Length, maxLength)];
        }

        return string.Join(
            "\n\n",
            chosen
                .OrderBy(x => x.Index)
                .Select(x => x.Text));
    }

    // ========================================================
    // TỪ KHÓA TRUY XUẤT
    // ========================================================

    private static HashSet<string> LayTuKhoa(
        string question)
    {
        string normalized = Normalize(question);

        HashSet<string> stopWords =
            new(StringComparer.Ordinal)
            {
                "tour", "tripi", "minh", "ban",
                "nay", "kia", "la", "co",
                "khong", "nhung", "nhu",
                "the", "nao", "cho", "voi",
                "duoc", "gi", "ve", "cua",
                "mot", "hay", "roi", "nhe",
                "di", "oi", "a", "nua"
            };

        return Regex.Matches(
                normalized,
                @"[a-z0-9]+")
            .Select(m => m.Value)
            .Where(word =>
                word.Length >= 3 &&
                !stopWords.Contains(word))
            .ToHashSet(StringComparer.Ordinal);
    }

    private static string Normalize(string text)
    {
        string decomposed = text
            .Trim()
            .ToLowerInvariant()
            .Replace('đ', 'd')
            .Normalize(NormalizationForm.FormD);

        var builder = new StringBuilder();

        foreach (char character in decomposed)
        {
            UnicodeCategory category =
                CharUnicodeInfo.GetUnicodeCategory(
                    character);

            if (category != UnicodeCategory.NonSpacingMark)
            {
                builder.Append(character);
            }
        }

        return builder.ToString()
            .Normalize(NormalizationForm.FormC);
    }

    // ========================================================
    // LỊCH SỬ CHỈ DÙNG LÀM NGỮ CẢNH TRÒ CHUYỆN
    // ========================================================

    private static string TaoLichSuChoRag(
        IReadOnlyList<TripiChatMessage>? history)
    {
        if (history == null || history.Count == 0)
        {
            return "(Chưa có lịch sử được cung cấp)";
        }

        var builder = new StringBuilder();

        foreach (TripiChatMessage item in history
            .Where(x =>
                x != null &&
                (x.Role == "user" ||
                 x.Role == "model") &&
                !string.IsNullOrWhiteSpace(x.Content))
            .TakeLast(MaxHistoryMessages))
        {
            string role =
                item.Role == "user"
                    ? "Khách"
                    : "Tripi";

            string content = item.Content.Trim();

            if (content.Length > 600)
            {
                content = content[..600];
            }

            string line =
                $"{role}: {content}\n";

            if (builder.Length + line.Length >
                MaxHistoryCharacters)
            {
                break;
            }

            builder.Append(line);
        }

        return builder.Length == 0
            ? "(Chưa có lịch sử phù hợp)"
            : builder.ToString();
    }

    // ========================================================
    // GỌI GOOGLE GEMINI
    // ========================================================

    private static async Task<JsonDocument>
        SendGeminiRequestAsync(
            string endpoint,
            string apiKey,
            object payload,
            CancellationToken cancellationToken)
    {
        for (int attempt = 1;
             attempt <= MaxAttempts;
             attempt++)
        {
            cancellationToken.ThrowIfCancellationRequested();

            using var request =
                new HttpRequestMessage(
                    HttpMethod.Post,
                    endpoint);

            request.Headers.Add(
                "x-goog-api-key",
                apiKey);

            request.Content =
                JsonContent.Create(payload);

            using HttpResponseMessage response =
                await Client.SendAsync(
                    request,
                    cancellationToken);

            if (response.IsSuccessStatusCode)
            {
                await using Stream stream =
                    await response.Content.ReadAsStreamAsync(
                        cancellationToken);

                return await JsonDocument.ParseAsync(
                    stream,
                    cancellationToken: cancellationToken);
            }

            bool retryable =
                response.StatusCode ==
                    HttpStatusCode.TooManyRequests ||
                response.StatusCode ==
                    HttpStatusCode.ServiceUnavailable;

            if (retryable && attempt < MaxAttempts)
            {
                await Task.Delay(
                    TimeSpan.FromSeconds(attempt * 2),
                    cancellationToken);

                continue;
            }

            throw new InvalidOperationException(
                $"Gemini RAG API lỗi HTTP " +
                $"{(int)response.StatusCode}.");
        }

        throw new InvalidOperationException(
            "Gemini đang quá tải. " +
            "Bạn có thể thử hỏi lại sau.");
    }

    // ========================================================
    // TRÍCH XUẤT TEXT TỪ GEMINI
    // ========================================================

    private static string ExtractText(
        JsonDocument document)
    {
        JsonElement root = document.RootElement;

        if (!root.TryGetProperty(
                "candidates",
                out JsonElement candidates) ||
            candidates.ValueKind !=
                JsonValueKind.Array ||
            candidates.GetArrayLength() == 0)
        {
            throw new InvalidOperationException(
                "Gemini chưa trả về câu trả lời RAG.");
        }

        JsonElement candidate = candidates[0];

        if (!candidate.TryGetProperty(
                "content",
                out JsonElement content) ||
            !content.TryGetProperty(
                "parts",
                out JsonElement parts) ||
            parts.ValueKind !=
                JsonValueKind.Array)
        {
            throw new InvalidOperationException(
                "Nội dung Gemini RAG không hợp lệ.");
        }

        string text = string.Join(
            "\n",
            parts.EnumerateArray()
                .Where(part =>
                    part.TryGetProperty(
                        "text",
                        out JsonElement value) &&
                    value.ValueKind ==
                        JsonValueKind.String)
                .Select(part =>
                    part.GetProperty("text").GetString())
                .Where(value =>
                    !string.IsNullOrWhiteSpace(value)));

        if (string.IsNullOrWhiteSpace(text))
        {
            throw new InvalidOperationException(
                "Gemini chưa tạo được nội dung RAG.");
        }

        return text.Trim();
    }

    private static string ReadString(
        JsonElement root,
        string propertyName)
    {
        if (!root.TryGetProperty(
                propertyName,
                out JsonElement value) ||
            value.ValueKind !=
                JsonValueKind.String)
        {
            return "";
        }

        return value.GetString()?.Trim() ?? "";
    }
